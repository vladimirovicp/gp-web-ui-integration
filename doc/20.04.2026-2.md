# Документация по API GPO в `chain/gpo.js`

Этот документ описывает, как в текущей реализации проекта использовать API-команды:

- UI label: `gpo-get-current-value`, внутреннее значение: `gpo_get_current_value`
- UI label: `gpo-list-children`, внутреннее значение: `gpo_list_children`
- UI label: `gpo-set-policy`, внутреннее значение: `gpo_set_policy`
- UI label: `gpo-get-policy`, внутреннее значение: `gpo_get_policy`

Описание основано на фактическом коде в [chain/gpo.js](/home/fad/git/gp-web-ui-integration/chain/gpo.js:304) и [chain/js/app.js](/home/fad/git/gp-web-ui-integration/chain/js/app.js:3322).

## 1. Как это работает сейчас

В `chain/gpo.js` при нажатии на кнопку `Execute Command` формируется RPC-вызов:

```js
var command = command_widget.get_value()[0];
var method = command.replace(/^gpo_/, '');
var path = path_widget.get_value()[0] || '/';

var args = [path];
var options = {
  version: IPA.api_version
};

if (command === 'gpo_set_policy') {
  options.json = JSON.parse(json_data);
}

var rpc_command = rpc.command({
  entity: 'gpo',
  method: method,
  args: args,
  options: options,
  on_success: function(data) {
    var result_data = data.result.result || {};
    console.log(result_data);
  }
});
```

То есть фактически выполняются методы:

- `gpo_get_current_value` -> `method: 'get_current_value'`
- `gpo_list_children` -> `method: 'list_children'`
- `gpo_set_policy` -> `method: 'set_policy'`
- `gpo_get_policy` -> `method: 'get_policy'`

Во всех случаях:

- `entity` всегда равен `'gpo'`
- в `args` передаётся только один аргумент: `path`
- в `options` всегда передаётся `version: IPA.api_version`
- только для `set_policy` дополнительно передаётся `options.json`

## 2. Важная особенность текущей реализации

В модальном окне GPUI имя выбранной политики передаётся в `app.init()` как `policyName`:

```js
app.init({
  containerId: 'gp__container',
  policyName: policyName,
  path: '/'
});
```

Но в текущем коде `chain/js/app.js` при загрузке политики используется только `path`:

```js
loadMainPolicy(options.path || "/");
```

А в `getPolicy()` RPC-вызов тоже формируется только из `path`:

```js
rpc.command({
  entity: "gpo",
  method: "get_policy",
  args: [policyPath],
  options: {
    version: IPA.api_version
  }
});
```

Это значит:

- текущая реализация вызовов опирается на `path`
- `policyName` в сам RPC-вызов этих методов сейчас не подставляется
- если на серверной стороне требуется дополнительная идентификация политики, это не отражено в текущем фронтенд-коде

Ниже документация описывает именно текущее поведение фронтенда.

## 3. Базовый шаблон вызова из консоли

Все примеры ниже можно запускать в браузерной консоли на странице FreeIPA, где уже доступны модули `freeipa/rpc` и `freeipa/ipa`.

Универсальный шаблон:

```js
require(['freeipa/rpc', 'freeipa/ipa'], function(rpc, IPA) {
  var path = '/';

  var command = rpc.command({
    entity: 'gpo',
    method: 'get_policy',
    args: [path],
    options: {
      version: IPA.api_version
    },
    on_success: function(data) {
      console.log('RAW RESPONSE:', data);
      console.log('RESULT ONLY:', data && data.result ? data.result.result : null);
    },
    on_error: function(xhr, text_status, error_thrown) {
      console.error('RPC ERROR:', {
        xhr: xhr,
        text_status: text_status,
        error: error_thrown
      });
    }
  });

  command.execute();
});
```

Что здесь важно:

- `RAW RESPONSE` показывает полный ответ RPC
- `RESULT ONLY` показывает полезную часть, которую текущий UI берёт как `data.result.result`
- `path` можно менять в зависимости от нужной ветки политики

## 4. `gpo_get_current_value`

### Назначение

Получить текущее значение узла политики по указанному пути.

### Что передать

Нужно передать:

- `entity: 'gpo'`
- `method: 'get_current_value'`
- `args: [path]`
- `options.version: IPA.api_version`

Минимальный пример:

```js
require(['freeipa/rpc', 'freeipa/ipa'], function(rpc, IPA) {
  var path = '/';

  var command = rpc.command({
    entity: 'gpo',
    method: 'get_current_value',
    args: [path],
    options: {
      version: IPA.api_version
    },
    on_success: function(data) {
      var result = data && data.result ? data.result.result : null;
      console.log('[gpo_get_current_value] raw:', data);
      console.log('[gpo_get_current_value] result:', result);
    },
    on_error: function(xhr, text_status, error_thrown) {
      console.error('[gpo_get_current_value] error:', text_status, error_thrown || xhr);
    }
  });

  command.execute();
});
```

### Что делает код

1. Загружает `rpc` и `IPA`
2. Берёт путь `path`
3. Вызывает метод `get_current_value`
4. После успешного ответа извлекает `data.result.result`
5. Выводит и полный ответ, и полезные данные в консоль

### Ожидаемый формат ответа

Точный состав полей зависит от backend-реализации, но фронтенд ожидает ответ в такой форме:

```js
{
  result: {
    result: {
      // полезные данные по текущему значению
    },
    value: "...",
    summary: "..."
  }
}
```

Для вывода результата используется:

```js
var result = data.result.result;
console.log(result);
```

### Пример вывода в консоль

```js
[gpo_get_current_value] raw: {result: {...}}
[gpo_get_current_value] result: {
  value: true
}
```

Пример содержимого `result` условный. Реальные поля возвращает сервер.

## 5. `gpo-list-children`

### Назначение

Получить список дочерних элементов для указанного узла политики.

### Что передать

Нужно передать:

- `entity: 'gpo'`
- `method: 'list_children'`
- `args: [path]`
- `options.version: IPA.api_version`

Пример вызова:

```js
require(['freeipa/rpc', 'freeipa/ipa'], function(rpc, IPA) {
  var path = '/';

  var command = rpc.command({
    entity: 'gpo',
    method: 'list_children',
    args: [path],
    options: {
      version: IPA.api_version
    },
    on_success: function(data) {
      var result = data && data.result ? data.result.result : null;
      console.log('[gpo-list-children] raw:', data);
      console.log('[gpo-list-children] result:', result);
      if (Array.isArray(result)) {
        console.table(result);
      }
    },
    on_error: function(xhr, text_status, error_thrown) {
      console.error('[gpo-list-children] error:', text_status, error_thrown || xhr);
    }
  });

  command.execute();
});
```

### Что делает код

1. Передаёт путь интересующего узла
2. Вызывает `list_children`
3. Достаёт `data.result.result`
4. Печатает результат
5. Если сервер вернул массив, дополнительно печатает его через `console.table()`

### Пример вывода в консоль

```js
[gpo-list-children] raw: {result: {...}}
[gpo-list-children] result: [
  {name: 'Computer Configuration', path: '/computer'},
  {name: 'User Configuration', path: '/user'}
]
```

Если backend возвращает не массив, а объект, тогда результат просто будет выведен как объект.

## 6. `gpo-set-policy`

### Назначение

Установить или обновить данные политики по указанному пути.

### Что передать

Нужно передать:

- `entity: 'gpo'`
- `method: 'set_policy'`
- `args: [path]`
- `options.version: IPA.api_version`
- `options.json: <объект с данными политики>`

Ключевой момент:

- `json_data` в интерфейсе вводится как строка
- затем код делает `JSON.parse(json_data)`
- в RPC уходит уже объект в `options.json`

### Пример данных

Например, если вы хотите отправить такие данные:

```json
{
  "enabled": true,
  "value": 120,
  "title": "Example policy"
}
```

То консольный вызов будет таким:

```js
require(['freeipa/rpc', 'freeipa/ipa'], function(rpc, IPA) {
  var path = '/';
  var jsonData = {
    enabled: true,
    value: 120,
    title: 'Example policy'
  };

  var command = rpc.command({
    entity: 'gpo',
    method: 'set_policy',
    args: [path],
    options: {
      version: IPA.api_version,
      json: jsonData
    },
    on_success: function(data) {
      var result = data && data.result ? data.result.result : null;
      console.log('[gpo-set-policy] sent json:', jsonData);
      console.log('[gpo-set-policy] raw:', data);
      console.log('[gpo-set-policy] result:', result);
    },
    on_error: function(xhr, text_status, error_thrown) {
      console.error('[gpo-set-policy] error:', text_status, error_thrown || xhr);
    }
  });

  command.execute();
});
```

### Вариант, полностью повторяющий логику UI

Этот пример показывает тот же подход, что и в `chain/gpo.js`, когда JSON сначала приходит строкой:

```js
require(['freeipa/rpc', 'freeipa/ipa'], function(rpc, IPA) {
  var path = '/';
  var jsonText = '{"enabled":true,"value":120,"title":"Example policy"}';

  try {
    var jsonObject = JSON.parse(jsonText);

    var command = rpc.command({
      entity: 'gpo',
      method: 'set_policy',
      args: [path],
      options: {
        version: IPA.api_version,
        json: jsonObject
      },
      on_success: function(data) {
        console.log('[gpo-set-policy] raw:', data);
        console.log('[gpo-set-policy] result:', data && data.result ? data.result.result : null);
      },
      on_error: function(xhr, text_status, error_thrown) {
        console.error('[gpo-set-policy] error:', text_status, error_thrown || xhr);
      }
    });

    command.execute();
  } catch (e) {
    console.error('Invalid JSON data:', e.message);
  }
});
```

### Что будет при невалидном JSON

В текущем UI заложено такое поведение:

```js
try {
  var json_obj = JSON.parse(json_data);
  options.json = json_obj;
} catch (e) {
  IPA.notify('Invalid JSON data: ' + e.message, 'error');
  return;
}
```

То есть:

- если строка JSON невалидна, RPC-вызов вообще не будет выполнен
- пользователь увидит ошибку парсинга

### Пример вывода в консоль

```js
[gpo-set-policy] sent json: {enabled: true, value: 120, title: 'Example policy'}
[gpo-set-policy] raw: {result: {...}}
[gpo-set-policy] result: {
  success: true
}
```

Содержимое `result` зависит от backend.

## 7. `gpo-get-policy`

### Назначение

Получить полное состояние политики по указанному пути.

### Что передать

Нужно передать:

- `entity: 'gpo'`
- `method: 'get_policy'`
- `args: [path]`
- `options.version: IPA.api_version`

Пример вызова:

```js
require(['freeipa/rpc', 'freeipa/ipa'], function(rpc, IPA) {
  var path = '/';

  var command = rpc.command({
    entity: 'gpo',
    method: 'get_policy',
    args: [path],
    options: {
      version: IPA.api_version
    },
    on_success: function(data) {
      var result = data && data.result ? data.result.result : null;
      console.log('[gpo-get-policy] raw:', data);
      console.log('[gpo-get-policy] result:', result);
      console.log('[gpo-get-policy] JSON:', JSON.stringify(result, null, 2));
    },
    on_error: function(xhr, text_status, error_thrown) {
      console.error('[gpo-get-policy] error:', text_status, error_thrown || xhr);
    }
  });

  command.execute();
});
```

### Где это уже используется в проекте

В `chain/js/app.js` уже есть функция:

```js
function getPolicy(path, onSuccess, onError) {
  var command = rpc.command({
    entity: "gpo",
    method: "get_policy",
    args: [path],
    options: {
      version: IPA.api_version
    },
    on_success: function(data) {
      var result = data && data.result ? data.result.result || {} : {};
      if (typeof onSuccess === "function") {
        onSuccess(result);
      }
    }
  });

  command.execute();
}
```

То есть текущая прикладная логика уже считает, что:

- основной полезный ответ находится в `data.result.result`
- если `result` отсутствует, используется пустой объект

### Пример вывода в консоль

```js
[gpo-get-policy] raw: {result: {...}}
[gpo-get-policy] result: {
  path: '/',
  children: [],
  value: {
    enabled: true
  }
}
```

Это пример структуры. Реальный формат приходит с сервера.

|---|---|---|---|---|
| `gpo-get-current-value` | `gpo_get_current_value` | `get_current_value` | `[path]` | `{ version: IPA.api_version }` |
| `gpo-list-children` | `gpo_list_children` | `list_children` | `[path]` | `{ version: IPA.api_version }` |

## 10. Итог

В текущем фронтенде все четыре команды вызываются одинаково по схеме:

- выбирается `method`
- передаётся один аргумент `path`
- добавляется `version: IPA.api_version`
- для `set_policy` добавляется `options.json`
- полезный ответ берётся из `data.result.result`

Если нужна серверно-ориентированная документация с точной структурой ответа для каждого метода, её уже нужно сверять с backend-реализацией этих RPC-команд. В текущем репозитории со стороны фронтенда видно именно такую схему интеграции.
